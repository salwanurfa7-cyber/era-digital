  <script src="https://cdn.jsdelivr.net/npm/bootstrap@5.3.0/dist/js/bootstrap.bundle.min.js"></script>
  <script type="module">
    import { initializeApp } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-app.js";
    import { getFirestore, collection, doc, onSnapshot, setDoc, deleteDoc } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-firestore.js";
    import { getAuth, signInAnonymously, onAuthStateChanged } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-auth.js";

    const firebaseConfig = JSON.parse(__firebase_config);
    const app = initializeApp(firebaseConfig);
    const db = getFirestore(app);
    const auth = getAuth(app);
    const appId = typeof __app_id !== 'undefined' ? __app_id : 'digital-literacy-salwa-v1';

    let currentModules = [];
    let isAdminMode = false;
    let score = 0;
    let adminModal, materiModal, quizModal;

    window.addEventListener('DOMContentLoaded', () => {
      adminModal = new bootstrap.Modal(document.getElementById('adminModal'));
      materiModal = new bootstrap.Modal(document.getElementById('materiModal'));
      quizModal = new bootstrap.Modal(document.getElementById('quizModal'));
      lucide.createIcons();
    });

    onAuthStateChanged(auth, (user) => {
      if (user) loadData();
      else signInAnonymously(auth);
    });

    window.startApp = (mode) => {
      const name = document.getElementById('username').value;
      if (!name) return alert("Ketik namamu dulu!");
      document.getElementById('user-display').innerText = name;
      document.getElementById('login-screen').style.display = 'none';
      document.getElementById('main-app').style.display = 'block';
    };

    window.checkAdmin = () => {
      const pw = prompt("Sandi Admin:");
      if (pw === "salwa123") {
        isAdminMode = true;
        document.getElementById('nav-admin').classList.remove('d-none');
        window.startApp('admin');
        window.switchView('admin');
      } else alert("Salah!");
    };

    function loadData() {
      const colPath = collection(db, 'artifacts', appId, 'public', 'data', 'modules');
      onSnapshot(colPath, (snap) => {
        currentModules = snap.docs.map(d => ({ id: d.id, ...d.data() }));
        renderUI();
      });
    }

    function renderUI() {
      const userList = document.getElementById('module-list-user');
      const adminList = document.getElementById('module-list-admin');
      userList.innerHTML = adminList.innerHTML = '';
      document.getElementById('stat-count').innerText = currentModules.length;

      currentModules.sort((a,b) => a.bab.localeCompare(b.bab)).forEach(m => {
        userList.innerHTML += `
          <div class="col-12 col-md-6">
            <div class="module-card shadow-sm">
              <div class="img-container">
                <span class="badge-bab">${m.bab}</span>
                <img src="https://images.unsplash.com/photo-1550751827-4bd374c3f58b?w=400">
              </div>
              <div class="p-3">
                <h6 class="module-title">${m.title}</h6>
                <div class="btn-action-group">
                  <button class="btn-m btn-read" onclick="openMateri('${m.id}')">Baca & Video</button>
                  <button class="btn-m btn-quiz" onclick="openQuiz('${m.id}')">Kuis</button>
                </div>
              </div>
            </div>
          </div>`;

        adminList.innerHTML += `
          <div class="col-12">
            <div class="card border-0 shadow-sm rounded-4 p-3 mb-2">
              <div class="d-flex justify-content-between">
                <div><span class="badge bg-primary">${m.bab}</span><br><strong>${m.title}</strong></div>
                <div class="d-flex gap-2">
                  <button class="btn btn-sm btn-outline-warning" onclick="editMateri('${m.id}')"><i data-lucide="edit-3" size="16"></i></button>
                  <button class="btn btn-sm btn-outline-danger" onclick="deleteMateri('${m.id}')"><i data-lucide="trash-2" size="16"></i></button>
                </div>
              </div>
            </div>
          </div>`;
      });
      lucide.createIcons();
    }

    window.saveMateri = async () => {
      const id = document.getElementById('edit-id').value || crypto.randomUUID();
      const data = {
        bab: document.getElementById('adm-bab').value,
        title: document.getElementById('adm-title').value,
        video: document.getElementById('adm-video').value,
        content: document.getElementById('adm-content').value,
        q: document.getElementById('adm-q').value,
        a: [document.getElementById('adm-a0').value, document.getElementById('adm-a1').value, document.getElementById('adm-a2').value],
        c: parseInt(document.getElementById('adm-c').value) || 0
      };
      await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'modules', id), data);
      adminModal.hide();
    };

    window.editMateri = (id) => {
      const m = currentModules.find(x => x.id === id);
      document.getElementById('edit-id').value = m.id;
      document.getElementById('adm-bab').value = m.bab;
      document.getElementById('adm-title').value = m.title;
      document.getElementById('adm-video').value = m.video;
      document.getElementById('adm-content').value = m.content;
      document.getElementById('adm-q').value = m.q;
      m.a.forEach((ans, i) => document.getElementById('adm-a'+i).value = ans);
      document.getElementById('adm-c').value = m.c;
      adminModal.show();
    };

    window.deleteMateri = async (id) => { if(confirm("Hapus?")) await deleteDoc(doc(db, 'artifacts', appId, 'public', 'data', 'modules', id)); };

    window.openMateri = (id) => {
      const m = currentModules.find(x => x.id === id);
      document.getElementById('mTitle').innerText = m.title;
      document.getElementById('mBody').innerHTML = m.content;
      const vBox = document.getElementById('videoContainer');
      const frame = document.getElementById('ytFrame');
      if(m.video) { vBox.classList.remove('d-none'); frame.src = m.video; }
      else { vBox.classList.add('d-none'); frame.src = ""; }
      materiModal.show();
    };

    window.stopVideo = () => { document.getElementById('ytFrame').src = ""; };

    window.openQuiz = (id) => {
      const m = currentModules.find(x => x.id === id);
      document.getElementById('qText').innerText = m.q;
      const opts = document.getElementById('qOptions');
      opts.innerHTML = '';
      m.a.forEach((ans, i) => {
        const btn = document.createElement('div');
        btn.className = 'quiz-option';
        btn.innerText = ans;
        btn.onclick = () => {
          if(i === m.c) { btn.classList.add('correct'); score += 10; document.getElementById('stat-score').innerText = score; setTimeout(() => quizModal.hide(), 1000); }
          else btn.classList.add('wrong');
        };
        opts.appendChild(btn);
      });
      quizModal.show();
    };

    window.openAdminModal = () => { document.getElementById('edit-id').value = ""; adminModal.show(); };
    window.switchView = (v) => {
      document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
      document.getElementById('page-'+v).classList.add('active');
      document.querySelectorAll('.nav-btn').forEach(b => b.classList.remove('active'));
      document.getElementById('nav-'+v).classList.add('active');
    };
  </script>
